import time
from threading import Lock
from langchain_core.embeddings import Embeddings


class RefreshableEmbeddings(Embeddings):
    def __init__(self, embedder_factory, ttl_seconds: int = 30, refresh_margin: int = 5):
        self.embedder_factory = embedder_factory
        self.ttl_seconds = ttl_seconds
        self.refresh_margin = refresh_margin

        self._lock = Lock()
        self._embedder = None
        self._expires_at = 0.0

        self._refresh()

    def _refresh(self):
        self._embedder = self.embedder_factory()
        self._expires_at = time.time() + self.ttl_seconds

    def _get_embedder(self):
        now = time.time()

        if now < self._expires_at - self.refresh_margin:
            return self._embedder

        with self._lock:
            # double-check, żeby dwa requesty naraz nie odświeżały tokena
            if time.time() >= self._expires_at - self.refresh_margin:
                self._refresh()

        return self._embedder

    def embed_query(self, text: str) -> list[float]:
        return self._get_embedder().embed_query(text)

    def embed_documents(self, texts: list[str]) -> list[list[float]]:
        return self._get_embedder().embed_documents(texts)
