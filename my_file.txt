from pathlib import Path

for pyfile in Path(".").rglob("__init__.py"):
    lines = pyfile.read_text().splitlines()
    modified = False
    new_lines = []

    for line in lines:
        stripped = line.strip()
        if (
            (stripped.startswith("import ") or stripped.startswith("from "))
            and "# noqa" not in stripped
        ):
            line += "  # noqa: F401"
            modified = True
        new_lines.append(line)

    if modified:
        pyfile.write_text("\n".join(new_lines) + "\n")
