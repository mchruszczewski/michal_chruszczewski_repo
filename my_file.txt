import os
import ast
import json

def extract_defs_from_source(source, tree):
    defs = {}
    for node in ast.iter_child_nodes(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            name = node.name
            body = ast.get_source_segment(source, node)
            if body:
                defs[name] = body
    return defs

def process_repository(root_dir, output_path):
    module_name = os.path.basename(os.path.abspath(root_dir))

    with open(output_path, "w", encoding="utf-8") as f:
        f.write('{\n')
        f.write(f'"{module_name}": {{\n')

        first_dir = True
        for dirpath, _, filenames in os.walk(root_dir):
            rel_dir = os.path.relpath(dirpath, root_dir)
            if rel_dir == '.':
                rel_dir = ''  # root level
            rel_dir_key = rel_dir.replace("\\", "/")

            dir_content = []
            for filename in filenames:
                if filename.endswith('.py'):
                    full_path = os.path.join(dirpath, filename)
                    try:
                        with open(full_path, encoding='utf-8') as source_file:
                            source = source_file.read()
                            tree = ast.parse(source)
                            defs = extract_defs_from_source(source, tree)
                            if defs:
                                file_entry = f'"{filename}": {json.dumps(defs, ensure_ascii=False)}'
                                dir_content.append(file_entry)
                    except Exception:
                        continue  # Skip files with encoding or parsing errors

            if dir_content:
                if not first_dir:
                    f.write(',\n')
                first_dir = False
                f.write(f'"{rel_dir_key}": {{\n')
                f.write(",\n".join(dir_content))
                f.write('\n}')

        f.write('\n}\n')  # close module
        f.write('}\n')    # close root
