import os
import ast

def extract_functions_with_docstrings(root_folder):
    functions_dict = {}
    
    for dirpath, _, filenames in os.walk(root_folder):
        for filename in filenames:
            if filename.endswith(".py") and filename != "__init__.py":
                module_name = filename[:-3]
                module_path = os.path.join(dirpath, filename)

                try:
                    with open(module_path, "r", encoding="utf-8") as f:
                        tree = ast.parse(f.read(), filename=module_name)

                    functions_info = {
                        node.name: ast.get_docstring(node) if ast.get_docstring(node) else "Brak opisu"
                        for node in ast.walk(tree) if isinstance(node, ast.FunctionDef)
                    }

                    relative_path = os.path.relpath(dirpath, root_folder).replace(os.sep, ".")
                    full_module_name = f"{relative_path}.{module_name}" if relative_path != "." else module_name

                    functions_dict[full_module_name] = functions_info

                except Exception as e:
                    print(f"Błąd podczas parsowania modułu {module_name}: {e}")

    return functions_dict

root_folder = "ścieżka_do_folderu"
functions_data = extract_functions_with_docstrings(root_folder)

for module, functions in functions_data.items():
    print(f"Moduł: {module}")
    for func, doc in functions.items():
        print(f"  - {func}: {doc}\n")
