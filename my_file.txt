import os
import ast
import json

def extract_defs_from_source(source, tree):
    result = {}
    for node in ast.iter_child_nodes(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef, ast.ClassDef)):
            body = ast.get_source_segment(source, node)
            if body:
                result[node.name] = body
    return result

def nested_set(dic, keys, value):
    """Ustawia zagnieżdżony słownik po liście kluczy"""
    for key in keys[:-1]:
        dic = dic.setdefault(key, {})
    dic[keys[-1]] = value

def process_repository_to_nested_json(root_dir, output_file_path):
    structure = {}

    root_dir = os.path.abspath(root_dir)
    for dirpath, _, filenames in os.walk(root_dir):
        rel_path = os.path.relpath(dirpath, root_dir).replace("\\", "/")
        path_parts = [] if rel_path == "." else rel_path.split("/")
        top_level_module = path_parts[0] if path_parts else None

        for filename in filenames:
            if not filename.endswith(".py"):
                continue
            if filename.startswith("test_") or filename.endswith("_test.py"):
                continue

            full_path = os.path.join(dirpath, filename)
            try:
                with open(full_path, encoding="utf-8") as fsrc:
                    source = fsrc.read()
                tree = ast.parse(source)
                defs = extract_defs_from_source(source, tree)
                if defs:
                    nested_set(structure, path_parts + [filename], defs)
            except Exception:
                continue

    with open(output_file_path, "w", encoding="utf-8") as f:
        json.dump(structure, f, indent=2, ensure_ascii=False)
