elif isinstance(func_def, dict):
    for func_name, func_args in func_def.items():
        if func_name not in self._custom_funcs:
            raise ValueError(f"Nieznana funkcja: {func_name}")
        
        func = self._custom_funcs[func_name]
        output_col = f"{col}_{func_name}"

        def agg_func(x, f=func, args=func_args):
            call_args = {}
            for k, v in args.items():
                call_args[k] = self.df.loc[x.index, v] if self._is_column_reference(v) else v
            return f(x, **call_args)

        agg_dict[output_col] = (col, agg_func)
