BBox = Tuple[float, float, float, float]  # (x0, top, x1, bottom)

def fix_spaced_text(text: str) -> str:
    # "M o d e l" -> "Model"
    text = re.sub(r'(?<=\b[A-Za-z])\s+(?=[A-Za-z]\b)', '', text)
    # ogarnij wielokrotne spacje
    text = re.sub(r'\s{2,}', ' ', text)
    return text.strip()


def extract_text_excluding_regions(page: pdfplumber.page.Page, regions: List[BBox]) -> str:
    words = page.extract_words(
        keep_blank_chars=False,
        use_text_flow=True,
        extra_attrs=["x0", "top", "x1", "bottom"],
    )

    kept = []
    for w in words:
        wbox: BBox = (float(w["x0"]), float(w["top"]), float(w["x1"]), float(w["bottom"]))
        if any(bbox_intersects(wbox, r) for r in regions):
            continue
        kept.append(w)

    # rebuild lines
    line_tol = 3.0
    kept.sort(key=lambda x: (round(float(x["top"]) / line_tol), float(x["x0"])))

    lines: Dict[int, List[str]] = {}
    for w in kept:
        k = round(float(w["top"]) / line_tol)
        lines.setdefault(k, []).append(w["text"])

    return "\n".join(" ".join(tokens) for _, tokens in sorted(lines.items())).strip()


def clean_cell(x: Any) -> str:
    if x is None:
        return ""
    s = str(x).strip()
    # spłaszcz whitespace (w tym \n w komórkach)
    return " ".join(s.split())


def make_headers(first_row: List[Any]) -> List[str]:
    headers = [clean_cell(h) for h in first_row]
    return [h if h else f"col_{i}" for i, h in enumerate(headers)]


def render_row_as_kv(headers: List[str], row: List[Any]) -> str:
    parts = []
    for h, v in zip(headers, row):
        cv = clean_cell(v)
        if cv:
            parts.append(f"{h}: {cv}")
    return "\n".join(parts).strip()


def pdf_to_contents(pdf_path: str) -> List[str]:
    """
    Returns pure content chunks:
    - table rows rendered as 'column: value' (one row -> one string)
    - page text with table areas removed (one page -> one string, if non-empty)
    """
    contents: List[str] = []

    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            # tables + bboxes
            tables = page.find_tables()
            table_regions: List[BBox] = []

            for t in tables:
                # t.bbox is a tuple
                table_regions.append(tuple(map(float, t.bbox)))  # type: ignore[arg-type]

                rows = t.extract() or []
                if len(rows) < 2:
                    continue

                headers = make_headers(rows[0])
                for row in rows[1:]:
                    txt = render_row_as_kv(headers, row)
                    if txt:
                        contents.append(txt)

            # text outside tables
            page_text = extract_text_excluding_regions(page, table_regions)
            if page_text:
                contents.append(page_text)

    return contents
