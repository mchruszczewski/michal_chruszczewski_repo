from langchain.schema import Document

def bq_retriever(query_embedding, k=5):
    sql = """
    SELECT content
    FROM VECTOR_SEARCH(
      TABLE dataset.embeddings,
      'embedding',
      @query_embedding,
      top_k => @k,
      distance_type => 'COSINE'
    )
    """
    job = client.query(
        sql,
        job_config=bigquery.QueryJobConfig(
            query_parameters=[
                bigquery.ArrayQueryParameter(
                    "query_embedding", "FLOAT64", query_embedding
                ),
                bigquery.ScalarQueryParameter("k", "INT64", k),
            ]
        ),
    )
    return [Document(page_content=row.content) for row in job]
