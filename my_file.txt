import pdfplumber
import xml.etree.ElementTree as ET

pdf_path = 'przyklad.pdf'
xml_path = 'wynik.xml'

root = ET.Element('document')

with pdfplumber.open(pdf_path) as pdf:
    for page_num, page in enumerate(pdf.pages, start=1):
        page_elem = ET.SubElement(root, 'page', attrib={'number': str(page_num)})

        # Wyodrębnianie tekstu
        text = page.extract_text()
        if text:
            text_elem = ET.SubElement(page_elem, 'text')
            text_elem.text = text

        # Wyodrębnianie tabel
        tables = page.extract_tables()
        for t_idx, table in enumerate(tables, start=1):
            table_elem = ET.SubElement(page_elem, 'table', attrib={'number': str(t_idx)})
            for row in table:
                row_elem = ET.SubElement(table_elem, 'row')
                for cell in row:
                    cell_elem = ET.SubElement(row_elem, 'cell')
                    cell_elem.text = cell if cell else ''

tree = ET.ElementTree(root)
tree.write(xml_path, encoding='utf-8', xml_declaration=True)

print(f'Plik XML został zapisany jako: {xml_path}')
