from pathlib import Path

for pyfile in Path(".").rglob("__init__.py"):
    try:
        lines = pyfile.read_text(encoding="utf-8").splitlines()
    except UnicodeDecodeError:
        print(f"Pomijam plik (problem z kodowaniem): {pyfile}")
        continue

    modified = False
    new_lines = []

    for line in lines:
        stripped = line.strip()
        if (
            (stripped.startswith("import ") or stripped.startswith("from "))
            and "# noqa" not in stripped
        ):
            line += "  # noqa: F401"
            modified = True
        new_lines.append(line)

    if modified:
        pyfile.write_text("\n".join(new_lines) + "\n", encoding="utf-8")
        print(f"Zmodyfikowano: {pyfile}")
