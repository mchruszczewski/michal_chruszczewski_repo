from typing import List, Tuple
import pandas as pd
from openpyxl import load_workbook


def parse_excel_tables(
    path: str,
    sheet_name=0,
    header_levels: int = 3,
) -> List[pd.DataFrame]:

    # 1️⃣ Wczytaj arkusz
    wb = load_workbook(path, data_only=True)
    ws = wb[sheet_name] if isinstance(sheet_name, str) else wb.worksheets[sheet_name]

    # 2️⃣ Rozszyj scalone komórki
    for merged_range in list(ws.merged_cells.ranges):
        min_row, min_col, max_row, max_col = merged_range.bounds
        value = ws.cell(min_row, min_col).value

        for r in range(min_row, max_row + 1):
            for c in range(min_col, max_col + 1):
                ws.cell(r, c).value = value

        ws.unmerge_cells(str(merged_range))

    # 3️⃣ Zamień na DataFrame
    df_raw = pd.DataFrame(ws.values)

    # usuń całkowicie puste kolumny
    df_raw = df_raw.dropna(axis=1, how="all")

    # 4️⃣ Wykryj bloki (czyli oddzielne tabele)
    non_empty = ~df_raw.isna().all(axis=1)

    blocks: List[Tuple[int, int]] = []
    start = None

    for i, has_data in enumerate(non_empty):
        if has_data and start is None:
            start = i
        elif not has_data and start is not None:
            blocks.append((start, i - 1))
            start = None

    if start is not None:
        blocks.append((start, len(df_raw) - 1))

    tables = []

    # 5️⃣ Parsuj każdą tabelę
    for start, end in blocks:
        sub = df_raw.iloc[start:end + 1].reset_index(drop=True)

        if len(sub) <= header_levels:
            continue  # za mała, pomijamy

        header = sub.iloc[:header_levels].fillna(method="ffill", axis=1)
        data = sub.iloc[header_levels:].reset_index(drop=True)

        # MultiIndex kolumn
        columns = pd.MultiIndex.from_frame(header.T.reset_index(drop=True))
        data.columns = columns

        tables.append(data)

    return tables
