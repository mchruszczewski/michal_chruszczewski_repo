import os
import importlib.util
import inspect

# Ścieżka do głównego folderu z modułami
root_folder = "ścieżka_do_folderu"  # Podmień na rzeczywistą ścieżkę

# Słownik do przechowywania wyników
functions_dict = {}

# Przechodzenie rekurencyjnie przez foldery i pliki
for dirpath, _, filenames in os.walk(root_folder):
    for filename in filenames:
        if filename.endswith(".py") and filename != "__init__.py":
            module_name = filename[:-3]  # Usunięcie rozszerzenia ".py"
            module_path = os.path.join(dirpath, filename)

            # Załaduj moduł dynamicznie
            spec = importlib.util.spec_from_file_location(module_name, module_path)
            module = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(module)

            # Pobierz wszystkie funkcje i ich opisy
            functions_info = {}
            for name, obj in inspect.getmembers(module, inspect.isfunction):
                docstring = inspect.getdoc(obj)  # Pobranie opisu funkcji
                functions_info[name] = docstring if docstring else "Brak opisu"

            # Zapisz wynik z uwzględnieniem ścieżki folderu jako namespace
            relative_path = os.path.relpath(dirpath, root_folder).replace(os.sep, ".")
            full_module_name = f"{relative_path}.{module_name}" if relative_path != "." else module_name
            functions_dict[full_module_name] = functions_info

# Wyświetlenie nazw funkcji i ich opisów
for module, functions in functions_dict.items():
    print(f"Moduł: {module}")
    for func, doc in functions.items():
        print(f"  - {func}: {doc}\n")
