import queue
import threading
from concurrent.futures import ThreadPoolExecutor
from typing import Iterable, List, Tuple

SENTINEL = object()

def batched(it, n):
    batch = []
    for x in it:
        batch.append(x)
        if len(batch) >= n:
            yield batch
            batch = []
    if batch:
        yield batch

def threaded_ingest_stream(
    chunks: Iterable, api_client, vectorstore,
    batch_size: int = 64, workers: int = 8, qsize: int = 32
):
    in_q: queue.Queue = queue.Queue(maxsize=qsize)     # backpressure
    out_q: queue.Queue = queue.Queue(maxsize=qsize)

    def producer():
        for batch in batched(chunks, batch_size):
            texts = [c.text for c in batch]
            metas = [c.metadata for c in batch]
            in_q.put((texts, metas))
        for _ in range(workers):
            in_q.put(SENTINEL)

    def api_worker():
        while True:
            item = in_q.get()
            if item is SENTINEL:
                out_q.put(SENTINEL)
                return
            texts, metas = item
            vectors = api_client.embed(texts)  # I/O
            out_q.put((texts, metas, vectors))

    def writer():
        finished = 0
        while finished < workers:
            item = out_q.get()
            if item is SENTINEL:
                finished += 1
                continue
            texts, metas, vectors = item
            # zależnie od vectorstore: add_embeddings / add_texts + embedding fn
            vectorstore.add_texts(texts=texts, metadatas=metas, embeddings=vectors)

    t_prod = threading.Thread(target=producer, daemon=True)
    t_write = threading.Thread(target=writer, daemon=True)
    t_prod.start()
    t_write.start()

    with ThreadPoolExecutor(max_workers=workers) as ex:
        futures = [ex.submit(api_worker) for _ in range(workers)]
        for f in futures:
            f.result()

    t_prod.join()
    t_write.join()
