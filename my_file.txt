import inspect

@classmethod
def get_mrn_defined_agg_funcs(cls) -> dict:
    """
    Zwraca słownik funkcji agregujących zdefiniowanych w klasie,
    które zaczynają się od 'calculate_'.

    Każda funkcja jest opisana poprzez:
    - docstring (jako 'doc')
    - listę argumentów (jako 'args' w formacie 'a, b=0, ...')
    """
    funcs = {}

    for name in dir(cls):
        if not name.startswith("calculate_"):
            continue

        attr = getattr(cls, name)
        if not callable(attr):
            continue

        sig = inspect.signature(attr)
        arg_list = ", ".join(str(param) for param in sig.parameters.values())

        funcs[name] = {
            "doc": (attr.__doc__ or "").strip(),
            "args": arg_list
        }

    return funcs
